## Standard Code Template

```go
// Generated by AI Agent on {{.Timestamp}}
// Task: {{.TaskName}}
// Tags: {{.Tags}}
// Reusable: true

package main

import (
    "crypto/sha256"
    "encoding/json"
    "flag"
    "fmt"
    "io"
    "log"
    "os"
    "path/filepath"
    "time"
)

// Config holds task configuration
type Config struct {
    TargetDir   string
    DryRun      bool
    JSONOutput  bool
    LogFile     string
    HistoryFile string
}

// HistoryEntry tracks filesystem operations
type HistoryEntry struct {
    Timestamp      time.Time `json:"timestamp"`
    Operation      string    `json:"operation"`
    OldPath        string    `json:"oldPath,omitempty"`
    NewPath        string    `json:"newPath,omitempty"`
    ChecksumBefore string    `json:"checksumBefore,omitempty"`
    ChecksumAfter  string    `json:"checksumAfter,omitempty"`
    Success        bool      `json:"success"`
    ErrorMessage   string    `json:"errorMessage,omitempty"`
}

// Result represents task execution result
type Result struct {
    Success       bool            `json:"success"`
    FilesAffected int             `json:"filesAffected"`
    Operations    []HistoryEntry  `json:"operations"`
    Duration      time.Duration   `json:"duration"`
    ErrorMessage  string          `json:"errorMessage,omitempty"`
}

var history []HistoryEntry

func main() {
    config := parseArgs()
    
    if config.DryRun {
        fmt.Println("=== DRY RUN MODE ===")
        fmt.Println("No changes will be made to filesystem")
    }
    
    startTime := time.Now()
    result := run(config)
    result.Duration = time.Since(startTime)
    
    if config.JSONOutput {
        output, _ := json.MarshalIndent(result, "", "  ")
        fmt.Println(string(output))
    } else {
        if result.Success {
            fmt.Printf("Task completed successfully. Files affected: %d\n", result.FilesAffected)
        } else {
            fmt.Printf("Task failed: %s\n", result.ErrorMessage)
            os.Exit(1)
        }
    }
    
    // Save history
    saveHistory(config.HistoryFile, result.Operations)
}

func parseArgs() Config {
    config := Config{}
    
    flag.StringVar(&config.TargetDir, "dir", ".", "Target directory")
    flag.BoolVar(&config.DryRun, "dry-run", false, "Preview changes without executing")
    flag.BoolVar(&config.JSONOutput, "json", false, "Output as JSON")
    flag.StringVar(&config.LogFile, "log", "execution.log", "Log file path")
    flag.StringVar(&config.HistoryFile, "history", "history.json", "History file path")
    
    flag.Parse()
    
    return config
}

func run(config Config) Result {
    result := Result{Operations: history}
    
    // TODO: Implement task-specific logic here
    
    result.Success = true
    result.FilesAffected = len(history)
    return result
}

func logOperation(entry HistoryEntry) {
    history = append(history, entry)
}

func calculateChecksum(filePath string) (string, error) {
    file, err := os.Open(filePath)
    if err != nil {
        return "", err
    }
    defer file.Close()
    
    hash := sha256.New()
    if _, err := io.Copy(hash, file); err != nil {
        return "", err
    }
    
    return fmt.Sprintf("%x", hash.Sum(nil)), nil
}

func saveHistory(path string, entries []HistoryEntry) error {
    data, err := json.MarshalIndent(entries, "", "  ")
    if err != nil {
        return err
    }
    return os.WriteFile(path, data, 0644)
}

func validatePath(path string) error {
    absPath, err := filepath.Abs(path)
    if err != nil {
        return fmt.Errorf("invalid path: %w", err)
    }
    
    // Security: Ensure path doesn't escape target directory
    // Additional validation as needed
    _ = absPath
    
    return nil
}
```

```go
func ValidateSyntax(code string) error {
    fset := token.NewFileSet()
    _, err := parser.ParseFile(fset, "main.go", code, parser.AllErrors)
    if err != nil {
        return fmt.Errorf("syntax error: %w", err)
    }
    return nil
}
```

### Step 2: Compilation Test

```go
func TestCompilation(code string, workDir string) error {
    // Write code to temp file
    mainPath := filepath.Join(workDir, "main.go")
    if err := os.WriteFile(mainPath, []byte(code), 0644); err != nil {
        return err
    }
    
    // Initialize go.mod
    modPath := filepath.Join(workDir, "go.mod")
    modContent := "module task\n\ngo 1.21\n"
    if err := os.WriteFile(modPath, []byte(modContent), 0644); err != nil {
        return err
    }
    
    // Test compilation
    cmd := exec.Command("go", "build", "-o", "/dev/null", ".")
    cmd.Dir = workDir
    output, err := cmd.CombinedOutput()
    if err != nil {
        return fmt.Errorf("compilation failed: %s", string(output))
    }
    
    return nil
}
```

## Self-Correction Loop

When compilation fails, AI attempts to fix:

```go
func GenerateWithRetry(request TaskRequest, maxRetries int) (*GeneratedCode, error) {
    var lastError error
    var code string
    
    for attempt := 0; attempt < maxRetries; attempt++ {
        if attempt == 0 {
            code = generateInitialCode(request)
        } else {
            code = fixCodeWithAI(code, lastError)
        }
        
        if err := ValidateSyntax(code); err != nil {
            lastError = err
            continue
        }
        
        if err := TestCompilation(code, request.WorkDir); err != nil {
            lastError = err
            continue
        }
        
        return &GeneratedCode{Code: code}, nil
    }
    
    return nil, fmt.Errorf("failed after %d attempts: %w", maxRetries, lastError)
}

func fixCodeWithAI(code string, err error) string {
    prompt := fmt.Sprintf(`
The following Golang code failed with error:

ERROR: %s

CODE:
%s

Fix the code to resolve the error. Return only the corrected Golang code.
`, err.Error(), code)
    
    return callLLM(prompt)
}
```
