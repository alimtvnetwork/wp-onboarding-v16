## Purpose

Define the standard Golang code templates used by the AI code generator. Templates ensure consistency, include required features (dry-run, history logging, checksums), and reduce generation errors.

---

## Template Categories

| Category | Use Case | Key Features |
|----------|----------|--------------|
| FileRename | Rename files by pattern | Path validation, checksums |
| FileDelete | Delete files with confirmation | Backup option, undo support |
| ContentTransform | Modify file contents | Read/write, pattern matching |
| IndexUpdate | Update index/overview files | Markdown parsing, cross-refs |
| BatchOperation | Multi-file operations | Progress reporting, parallelism |
| DataValidation | Validate file structure | Schema checking, error reports |

## Base Template

All generated code inherits from this base structure:

```go
// Generated by AI Agent on {{.Timestamp}}
// Task: {{.TaskName}}
// Tags: {{.Tags}}
// Reusable: {{.IsReusable}}
// Template: {{.TemplateType}}

package main

import (
    "crypto/sha256"
    "encoding/json"
    "flag"
    "fmt"
    "io"
    "log"
    "os"
    "path/filepath"
    "time"
)

// ========================================
// Configuration
// ========================================

type Config struct {
    TargetDir   string `json:"targetDir"`
    DryRun      bool   `json:"dryRun"`
    JSONOutput  bool   `json:"jsonOutput"`
    Verbose     bool   `json:"verbose"`
    LogFile     string `json:"logFile"`
    HistoryFile string `json:"historyFile"`
}

// ========================================
// History Tracking
// ========================================

type OperationType string

const (
    OpCreate OperationType = "create"
    OpRead   OperationType = "read"
    OpUpdate OperationType = "update"
    OpDelete OperationType = "delete"
    OpRename OperationType = "rename"
    OpMove   OperationType = "move"
    OpCopy   OperationType = "copy"
)

type HistoryEntry struct {
    Timestamp      time.Time     `json:"timestamp"`
    Operation      OperationType `json:"operation"`
    OldPath        string        `json:"oldPath,omitempty"`
    NewPath        string        `json:"newPath,omitempty"`
    FileSizeBytes  int64         `json:"fileSizeBytes,omitempty"`
    ChecksumBefore string        `json:"checksumBefore,omitempty"`
    ChecksumAfter  string        `json:"checksumAfter,omitempty"`
    Success        bool          `json:"success"`
    ErrorMessage   string        `json:"errorMessage,omitempty"`
}

// ========================================
// Result Types
// ========================================

type Result struct {
    Success       bool           `json:"success"`
    FilesAffected int            `json:"filesAffected"`
    Operations    []HistoryEntry `json:"operations"`
    Duration      time.Duration  `json:"duration"`
    ErrorMessage  string         `json:"errorMessage,omitempty"`
}

// ========================================
// Global State
// ========================================

var (
    history []HistoryEntry
    config  Config
)

// ========================================
// Main Entry Point
// ========================================

func main() {
    config = parseArgs()

    if config.DryRun {
        printHeader("DRY RUN MODE - No changes will be made")
    }

    startTime := time.Now()
    result := run(config)
    result.Duration = time.Since(startTime)

    outputResult(result, config)
    saveHistory(config.HistoryFile)

    if !result.Success {
        os.Exit(1)
    }
}

func parseArgs() Config {
    c := Config{}

    flag.StringVar(&c.TargetDir, "dir", ".", "Target directory")
    flag.BoolVar(&c.DryRun, "dry-run", false, "Preview changes without executing")
    flag.BoolVar(&c.JSONOutput, "json", false, "Output as JSON")
    flag.BoolVar(&c.Verbose, "verbose", false, "Verbose output")
    flag.StringVar(&c.LogFile, "log", "execution.log", "Log file path")
    flag.StringVar(&c.HistoryFile, "history", "history.json", "History file path")

    flag.Parse()
    return c
}

// ========================================
// Core Logic (Template-Specific)
// ========================================

func run(config Config) Result {
    result := Result{Operations: []HistoryEntry{}}

    // TODO: Template-specific implementation

    result.Success = true
    result.Operations = history
    result.FilesAffected = len(history)
    return result
}

// ========================================
// Utility Functions
// ========================================

func logOperation(entry HistoryEntry) {
    entry.Timestamp = time.Now()
    history = append(history, entry)

    if config.Verbose {
        fmt.Printf("[%s] %s: %s -> %s (success: %v)\n",
            entry.Operation, entry.Timestamp.Format(time.RFC3339),
            entry.OldPath, entry.NewPath, entry.Success)
    }
}

func calculateChecksum(filePath string) (string, error) {
    file, err := os.Open(filePath)
    if err != nil {
        return "", err
    }
    defer file.Close()

    hash := sha256.New()
    if _, err := io.Copy(hash, file); err != nil {
        return "", err
    }

    return fmt.Sprintf("%x", hash.Sum(nil)), nil
}

func validatePath(basePath, targetPath string) error {
    absBase, err := filepath.Abs(basePath)
    if err != nil {
        return fmt.Errorf("invalid base path: %w", err)
    }

    absTarget, err := filepath.Abs(targetPath)
    if err != nil {
        return fmt.Errorf("invalid target path: %w", err)
    }

    // Security: Ensure path doesn't escape base directory
    rel, err := filepath.Rel(absBase, absTarget)
    if err != nil || len(rel) > 2 && rel[:2] == ".." {
        return fmt.Errorf("path escapes base directory: %s", targetPath)
    }

    return nil
}

func saveHistory(path string) error {
    if len(history) == 0 {
        return nil
    }

    data, err := json.MarshalIndent(history, "", "  ")
    if err != nil {
        return err
    }
    return os.WriteFile(path, data, 0644)
}

func outputResult(result Result, config Config) {
    if config.JSONOutput {
        output, _ := json.MarshalIndent(result, "", "  ")
        fmt.Println(string(output))
    } else {
        if result.Success {
            fmt.Printf("✅ Task completed. Files affected: %d, Duration: %v\n",
                result.FilesAffected, result.Duration)
        } else {
            fmt.Printf("❌ Task failed: %s\n", result.ErrorMessage)
        }
    }
}

func printHeader(msg string) {
    fmt.Println("========================================")
    fmt.Println(msg)
    fmt.Println("========================================")
}
```

## Template: FileRename

Rename files matching a pattern.

```go
func run(config Config) Result {
    result := Result{Operations: []HistoryEntry{}}

    files, err := filepath.Glob(filepath.Join(config.TargetDir, "*"))
    if err != nil {
        result.ErrorMessage = fmt.Sprintf("failed to list files: %v", err)
        return result
    }

    for _, oldPath := range files {
        if err := validatePath(config.TargetDir, oldPath); err != nil {
            continue
        }

        // Calculate new name (template-specific logic)
        dir := filepath.Dir(oldPath)
        base := filepath.Base(oldPath)
        newBase := strings.ToLower(base) // Example: lowercase
        newPath := filepath.Join(dir, newBase)

        if oldPath == newPath {
            continue // No change needed
        }

        entry := HistoryEntry{
            Operation: OpRename,
            OldPath:   oldPath,
            NewPath:   newPath,
        }

        // Calculate checksum before
        if checksum, err := calculateChecksum(oldPath); err == nil {
            entry.ChecksumBefore = checksum
        }

        if config.DryRun {
            fmt.Printf("Would rename: %s -> %s\n", oldPath, newPath)
            entry.Success = true
        } else {
            if err := os.Rename(oldPath, newPath); err != nil {
                entry.Success = false
                entry.ErrorMessage = err.Error()
            } else {
                entry.Success = true
                // Checksum after (same for rename)
                entry.ChecksumAfter = entry.ChecksumBefore
            }
        }

        logOperation(entry)
    }

    result.Success = true
    result.Operations = history
    result.FilesAffected = len(history)
    return result
}
```

## Template: ContentTransform

Transform file contents using pattern matching.

```go
func run(config Config) Result {
    result := Result{Operations: []HistoryEntry{}}

    files, err := filepath.Glob(filepath.Join(config.TargetDir, "*.md"))
    if err != nil {
        result.ErrorMessage = fmt.Sprintf("failed to list files: %v", err)
        return result
    }

    for _, filePath := range files {
        entry := HistoryEntry{
            Operation: OpUpdate,
            OldPath:   filePath,
            NewPath:   filePath,
        }

        // Read current content
        content, err := os.ReadFile(filePath)
        if err != nil {
            entry.Success = false
            entry.ErrorMessage = err.Error()
            logOperation(entry)
            continue
        }

        // Calculate checksum before
        entry.ChecksumBefore, _ = calculateChecksum(filePath)
        entry.FileSizeBytes = int64(len(content))

        // Transform content (template-specific logic)
        newContent := transformContent(string(content))

        if string(content) == newContent {
            continue // No change needed
        }

        if config.DryRun {
            fmt.Printf("Would transform: %s\n", filePath)
            entry.Success = true
        } else {
            if err := os.WriteFile(filePath, []byte(newContent), 0644); err != nil {
                entry.Success = false
                entry.ErrorMessage = err.Error()
            } else {
                entry.Success = true
                entry.ChecksumAfter, _ = calculateChecksum(filePath)
            }
        }

        logOperation(entry)
    }

    result.Success = true
    result.Operations = history
    result.FilesAffected = len(history)
    return result
}

func transformContent(content string) string {
    // Template-specific transformation logic
    // Example: Update version numbers, fix links, etc.
    return content
}
```

## Template: IndexUpdate

Update index files with cross-references.

```go
type IndexEntry struct {
    Number      int    `json:"number"`
    Name        string `json:"name"`
    Path        string `json:"path"`
    Description string `json:"description"`
    Status      string `json:"status"`
}

func run(config Config) Result {
    result := Result{Operations: []HistoryEntry{}}

    // Scan directory for files
    entries, err := scanDirectory(config.TargetDir)
    if err != nil {
        result.ErrorMessage = err.Error()
        return result
    }

    // Generate index content
    indexContent := generateIndexContent(entries)

    // Write to 00-overview.md
    indexPath := filepath.Join(config.TargetDir, "00-overview.md")
    entry := HistoryEntry{
        Operation: OpUpdate,
        OldPath:   indexPath,
        NewPath:   indexPath,
    }

    if _, err := os.Stat(indexPath); err == nil {
        entry.ChecksumBefore, _ = calculateChecksum(indexPath)
    }

    if config.DryRun {
        fmt.Printf("Would update index: %s\n", indexPath)
        fmt.Printf("Entries found: %d\n", len(entries))
        entry.Success = true
    } else {
        if err := os.WriteFile(indexPath, []byte(indexContent), 0644); err != nil {
            entry.Success = false
            entry.ErrorMessage = err.Error()
        } else {
            entry.Success = true
            entry.ChecksumAfter, _ = calculateChecksum(indexPath)
        }
    }

    logOperation(entry)

    result.Success = true
    result.Operations = history
    result.FilesAffected = len(history)
    return result
}

func scanDirectory(dir string) ([]IndexEntry, error) {
    var entries []IndexEntry
    // Scan and parse directory contents
    return entries, nil
}

func generateIndexContent(entries []IndexEntry) string {
    var sb strings.Builder
    sb.WriteString("# Index\n\n")
    sb.WriteString("| # | File | Description | Status |\n")
    sb.WriteString("|---|------|-------------|--------|\n")

    for _, e := range entries {
        sb.WriteString(fmt.Sprintf("| %02d | [%s](%s) | %s | %s |\n",
            e.Number, e.Name, e.Path, e.Description, e.Status))
    }

    return sb.String()
}
```
