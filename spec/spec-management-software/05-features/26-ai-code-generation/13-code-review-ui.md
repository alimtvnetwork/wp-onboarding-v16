# 13. Code Review UI

**Version:** 1.0.0  
**Status:** Planned  
**Updated:** 2026-01-29  
**Parent:** [Overview](./00-overview.md)

## Purpose

Define the frontend interface for reviewing, editing, and approving AI-generated Golang code before execution. Provides syntax highlighting, diff views, and dry-run result visualization.

### Code Viewer Component

```tsx
import { Prism as SyntaxHighlighter } from 'react-syntax-highlighter';
import { vscDarkPlus } from 'react-syntax-highlighter/dist/esm/styles/prism';

interface CodeViewerProps {
  readonly code: string;
  readonly language: string;
  readonly showLineNumbers: boolean;
  readonly maxHeight?: string;
}

export function CodeViewer({
  code,
  language,
  showLineNumbers,
  maxHeight = "400px",
}: CodeViewerProps): JSX.Element {
  const handleCopy = async (): Promise<void> => {
    await navigator.clipboard.writeText(code);
    toast.success("Code copied to clipboard");
  };

  return (
    <div className="relative rounded-lg overflow-hidden border border-border">
      <div className="absolute top-2 right-2 z-10 flex gap-2">
        <Button variant="ghost" size="sm" onClick={handleCopy}>
          <Copy className="h-4 w-4" />
        </Button>
      </div>
      <ScrollArea style={{ maxHeight }}>
        <SyntaxHighlighter
          language={language}
          style={vscDarkPlus}
          showLineNumbers={showLineNumbers}
          customStyle={{
            margin: 0,
            borderRadius: 0,
            fontSize: "13px",
          }}
        >
          {code}
        </SyntaxHighlighter>
      </ScrollArea>
    </div>
  );
}
```

### Dry-Run Results Component

```tsx
interface DryRunResultsProps {
  readonly result: DryRunResult;
}

export function DryRunResults({ result }: DryRunResultsProps): JSX.Element {
  return (
    <div className="space-y-4">
      {/* Status summary */}
      <div className="flex items-center gap-4">
        <div className="flex items-center gap-2">
          {result.compilationOk ? (
            <CheckCircle className="h-5 w-5 text-green-500" />
          ) : (
            <XCircle className="h-5 w-5 text-red-500" />
          )}
          <span>Compilation: {result.compilationOk ? "Success" : "Failed"}</span>
        </div>
        <div className="flex items-center gap-2">
          <FileIcon className="h-5 w-5 text-muted-foreground" />
          <span>Files affected: {result.affectedFiles.length}</span>
        </div>
        <div className="flex items-center gap-2">
          <Clock className="h-5 w-5 text-muted-foreground" />
          <span>Est. time: {formatDuration(result.estimatedTime)}</span>
        </div>
      </div>

      {/* Compilation errors */}
      {result.compileErrors.length > 0 && (
        <Alert variant="destructive">
          <AlertTitle>Compilation Errors</AlertTitle>
          <AlertDescription>
            <ul className="list-disc pl-4 mt-2">
              {result.compileErrors.map((err, i) => (
                <li key={i} className="font-mono text-sm">
                  Line {err.line}: {err.message}
                </li>
              ))}
            </ul>
          </AlertDescription>
        </Alert>
      )}

      {/* Planned operations */}
      {result.operations.length > 0 && (
        <div>
          <h4 className="font-medium mb-2">Planned Operations</h4>
          <div className="border rounded-lg divide-y">
            {result.operations.slice(0, 5).map((op, i) => (
              <div key={i} className="p-2 flex items-center gap-2 text-sm">
                <span className="text-muted-foreground">{op.order}.</span>
                <OperationIcon operation={op.operation} />
                <span className="font-mono">{op.oldPath}</span>
                {op.newPath && (
                  <>
                    <ArrowRight className="h-4 w-4" />
                    <span className="font-mono">{op.newPath}</span>
                  </>
                )}
              </div>
            ))}
            {result.operations.length > 5 && (
              <div className="p-2 text-sm text-muted-foreground">
                ... and {result.operations.length - 5} more
              </div>
            )}
          </div>
        </div>
      )}

      {/* Warnings */}
      {result.warnings.length > 0 && (
        <Alert>
          <AlertTriangle className="h-4 w-4" />
          <AlertTitle>Warnings</AlertTitle>
          <AlertDescription>
            <ul className="list-disc pl-4 mt-2">
              {result.warnings.map((warning, i) => (
                <li key={i}>{warning}</li>
              ))}
            </ul>
          </AlertDescription>
        </Alert>
      )}
    </div>
  );
}
```

## State Management

```typescript
interface CodeReviewState {
  readonly request: ApprovalRequest | null;
  readonly isOpen: boolean;
  readonly isLoading: boolean;
  readonly isEditing: boolean;
  readonly editedCode: string | null;
  readonly dryRunResult: DryRunResult | null;
  readonly error: string | null;
}

enum CodeReviewAction {
  OpenDialog = "open_dialog",
  CloseDialog = "close_dialog",
  StartEdit = "start_edit",
  SaveEdit = "save_edit",
  CancelEdit = "cancel_edit",
  RunDryRun = "run_dry_run",
  DryRunComplete = "dry_run_complete",
  Approve = "approve",
  Reject = "reject",
  SetError = "set_error",
}

function codeReviewReducer(
  state: CodeReviewState,
  action: { type: CodeReviewAction; payload?: unknown }
): CodeReviewState {
  switch (action.type) {
    case CodeReviewAction.OpenDialog:
      return {
        ...state,
        isOpen: true,
        request: action.payload as ApprovalRequest,
        isEditing: false,
        editedCode: null,
      };
    case CodeReviewAction.CloseDialog:
      return {
        ...state,
        isOpen: false,
        request: null,
        dryRunResult: null,
      };
    case CodeReviewAction.StartEdit:
      return {
        ...state,
        isEditing: true,
        editedCode: state.request?.golangCode ?? null,
      };
    case CodeReviewAction.SaveEdit:
      return {
        ...state,
        isEditing: false,
        dryRunResult: null, // Requires new dry-run
      };
    // ... other cases
    default:
      const _exhaustive: never = action.type;
      throw new Error(`Unhandled action: ${_exhaustive}`);
  }
}
```

## API Integration

```typescript
async function runDryRun(requestId: string): Promise<DryRunResult> {
  const response = await fetch(
    `/api/v1/code-generation/dry-run/${requestId}`,
    { method: "POST" }
  );
  
  if (!response.ok) {
    throw new Error("Dry-run failed");
  }
  
  return response.json();
}

async function approveRequest(
  requestId: string,
  userId: string
): Promise<ApprovalRequest> {
  const response = await fetch("/api/v1/code-generation/approve", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      requestId,
      action: "approve",
      userId,
    }),
  });
  
  if (!response.ok) {
    throw new Error("Approval failed");
  }
  
  return response.json();
}
```
