## Purpose

Provides integration with Yoast SEO plugin to detect missing focus keywords, auto-generate keywords from titles, manage multiple keywords (Premium), and optimize meta descriptions with configurable trimming.

## YoastSettingsService

### Settings Loading Priority

```
1. Check SQLite YoastSettings table
2. If not exists OR SeedVersion < ConfigVersion:
   - Load from config/yoast-defaults.json
   - Insert/Update in SQLite (if IsUserModified = false)
3. Return merged settings
```

### Interface

```php
interface YoastSettingsServiceInterface
{
    public function getSetting(string $key): mixed;
    public function setSetting(string $key, mixed $value): void;
    public function getAllSettings(): array;
    public function resetToDefaults(): void;
    public function exportSettings(): array;
    public function importSettings(array $settings): ValidationResult;
    public function seedFromConfig(): void;
}
```

### Implementation

```php
class YoastSettingsService implements YoastSettingsServiceInterface
{
    private const CONFIG_FILE = 'config/yoast-defaults.json';
    
    public function __construct(
        private readonly YoastSettingsRepository $repository,
        private readonly ConfigLoader $configLoader
    ) {}
    
    public function getSetting(string $key): mixed
    {
        $setting = $this->repository->findByKey($key);
        
        if (!$setting) {
            return $this->getDefaultValue($key);
        }
        
        return $setting->getValue();
    }
    
    public function setSetting(string $key, mixed $value): void
    {
        $setting = $this->repository->findByKey($key) ?? new YoastSettingsEntity();
        $setting->settingKey = $key;
        $setting->settingValue = is_array($value) ? json_encode($value) : (string) $value;
        $setting->settingType = $this->detectType($value);
        $setting->isUserModified = true;
        $setting->updatedAt = new DateTimeImmutable();
        
        $this->repository->save($setting);
    }
    
    public function seedFromConfig(): void
    {
        $config = $this->configLoader->load(self::CONFIG_FILE);
        $configVersion = $config['version'];
        
        foreach ($this->flattenSettings($config['settings']) as $key => $value) {
            $existing = $this->repository->findByKey($key);
            
            // Only seed if not exists OR version newer AND not user-modified
            if (!$existing || 
                (version_compare($configVersion, $existing->seedVersion ?? '0.0.0', '>') 
                 && !$existing->isUserModified)) {
                $this->repository->upsert($key, $value, $configVersion);
            }
        }
    }
}
```

## FocusKeywordService

### Generate from Title

```php
class FocusKeywordService
{
    public function generateFromTitle(string $title, array $settings): string
    {
        $keyword = $title;
        
        // Remove stop words if configured
        if ($settings['exclude_stop_words']) {
            $keyword = $this->removeStopWords($keyword, $settings['stop_words']);
        }
        
        // Trim to max words
        $words = explode(' ', $keyword);
        if (count($words) > $settings['max_words']) {
            $words = array_slice($words, 0, $settings['max_words']);
            $keyword = implode(' ', $words);
        }
        
        // Trim to max length
        if (strlen($keyword) > $settings['max_length']) {
            $keyword = $this->trimToLength($keyword, $settings['max_length'], $settings['trim_mode']);
        }
        
        return trim($keyword);
    }
    
    private function trimToLength(string $text, int $maxLength, string $mode): string
    {
        if ($mode === 'word_boundary') {
            // Trim at word boundary
            $trimmed = substr($text, 0, $maxLength);
            $lastSpace = strrpos($trimmed, ' ');
            return $lastSpace !== false ? substr($trimmed, 0, $lastSpace) : $trimmed;
        }
        
        // Hard cut
        return substr($text, 0, $maxLength);
    }
    
    public function setFocusKeyword(int $postId, string $keyword): bool
    {
        return update_post_meta($postId, '_yoast_wpseo_focuskw', $keyword);
    }
    
    public function getFocusKeyword(int $postId): ?string
    {
        return get_post_meta($postId, '_yoast_wpseo_focuskw', true) ?: null;
    }
}
```

## MultipleKeywordsService (Premium)

### Extract Keywords from Title

```php
class MultipleKeywordsService
{
    public function __construct(
        private readonly YoastDetector $detector,
        private readonly YoastSettingsService $settings
    ) {}
    
    public function extractKeywords(string $title, array $settings): array
    {
        if (!$this->detector->isYoastPremiumActive()) {
            throw new YoastPremiumRequiredException(14950);
        }
        
        $words = $this->tokenize($title);
        $keywords = [];
        
        foreach ($words as $word) {
            if (strlen($word) < $settings['min_word_length']) {
                continue;
            }
            if ($settings['exclude_numbers'] && is_numeric($word)) {
                continue;
            }
            $keywords[] = strtolower($word);
        }
        
        // Limit to max keywords
        return array_slice(array_unique($keywords), 0, $settings['max_keywords']);
    }
    
    public function setMultipleKeywords(int $postId, array $keywords): bool
    {
        $formatted = array_map(fn($kw) => ['keyword' => $kw, 'score' => 0], $keywords);
        return update_post_meta($postId, '_yoast_wpseo_focuskeywords', json_encode($formatted));
    }
    
    public function getMultipleKeywords(int $postId): array
    {
        $raw = get_post_meta($postId, '_yoast_wpseo_focuskeywords', true);
        return $raw ? json_decode($raw, true) : [];
    }
}
```

## MetaDescriptionService

### Trim Modes

| Mode | Behavior |
|------|----------|
| `hard_cut` | Cut exactly at max_length |
| `remove_last_word` | Remove last word until under limit |
| `sentence_boundary` | Cut at last complete sentence |
| `smart_trim` | Remove last word, add ellipsis if needed |

### Implementation

```php
class MetaDescriptionService
{
    public function trimDescription(string $description, array $settings): string
    {
        $maxLength = $settings['max_length'];
        
        if (strlen($description) <= $maxLength) {
            return $description;
        }
        
        return match($settings['trim_mode']) {
            'hard_cut' => $this->hardCut($description, $maxLength, $settings['add_ellipsis']),
            'remove_last_word' => $this->removeLastWord($description, $maxLength, $settings['add_ellipsis']),
            'sentence_boundary' => $this->sentenceBoundary($description, $maxLength),
            'smart_trim' => $this->smartTrim($description, $maxLength),
            default => $this->removeLastWord($description, $maxLength, $settings['add_ellipsis'])
        };
    }
    
    private function hardCut(string $text, int $maxLength, bool $addEllipsis): string
    {
        $suffix = $addEllipsis ? '...' : '';
        $cutLength = $maxLength - strlen($suffix);
        return substr($text, 0, $cutLength) . $suffix;
    }
    
    private function removeLastWord(string $text, int $maxLength, bool $addEllipsis): string
    {
        $suffix = $addEllipsis ? '...' : '';
        $targetLength = $maxLength - strlen($suffix);
        
        while (strlen($text) > $targetLength) {
            $lastSpace = strrpos($text, ' ');
            if ($lastSpace === false) {
                return substr($text, 0, $targetLength) . $suffix;
            }
            $text = substr($text, 0, $lastSpace);
        }
        
        return $text . $suffix;
    }
    
    private function sentenceBoundary(string $text, int $maxLength): string
    {
        // Find last sentence ending before maxLength
        $truncated = substr($text, 0, $maxLength);
        $lastPeriod = strrpos($truncated, '.');
        $lastQuestion = strrpos($truncated, '?');
        $lastExclaim = strrpos($truncated, '!');
        
        $lastSentence = max($lastPeriod ?: 0, $lastQuestion ?: 0, $lastExclaim ?: 0);
        
        if ($lastSentence > $maxLength * 0.5) {
            return substr($text, 0, $lastSentence + 1);
        }
        
        return $this->removeLastWord($text, $maxLength, true);
    }
    
    private function smartTrim(string $text, int $maxLength): string
    {
        // Try sentence boundary first, fall back to remove_last_word
        $result = $this->sentenceBoundary($text, $maxLength);
        if (strlen($result) >= $maxLength * 0.6) {
            return $result;
        }
        return $this->removeLastWord($text, $maxLength, true);
    }
    
    public function setMetaDescription(int $postId, string $description): bool
    {
        return update_post_meta($postId, '_yoast_wpseo_metadesc', $description);
    }
    
    public function getMetaDescription(int $postId): ?string
    {
        return get_post_meta($postId, '_yoast_wpseo_metadesc', true) ?: null;
    }
}
```

## YoastContentAnalyzer

### Find Missing SEO Data

```php
class YoastContentAnalyzer
{
    public function findMissingFocusKeywords(array $filters = []): array
    {
        global $wpdb;
        
        $postTypes = $filters['post_types'] ?? ['post', 'page'];
        $postTypesSql = "'" . implode("','", array_map('esc_sql', $postTypes)) . "'";
        
        $query = "
            SELECT p.ID, p.post_title, p.post_type, p.post_status
            FROM {$wpdb->posts} p
            LEFT JOIN {$wpdb->postmeta} pm ON p.ID = pm.post_id 
                AND pm.meta_key = '_yoast_wpseo_focuskw'
            WHERE p.post_type IN ({$postTypesSql})
                AND p.post_status = 'publish'
                AND (pm.meta_value IS NULL OR pm.meta_value = '')
            ORDER BY p.post_date DESC
        ";
        
        if (isset($filters['limit'])) {
            $query .= " LIMIT " . (int) $filters['limit'];
        }
        
        return $wpdb->get_results($query, ARRAY_A);
    }
    
    public function findOversizedMetaDescriptions(int $maxLength, array $filters = []): array
    {
        global $wpdb;
        
        $postTypes = $filters['post_types'] ?? ['post', 'page'];
        $postTypesSql = "'" . implode("','", array_map('esc_sql', $postTypes)) . "'";
        
        return $wpdb->get_results($wpdb->prepare("
            SELECT p.ID, p.post_title, p.post_type, pm.meta_value as meta_description,
                   LENGTH(pm.meta_value) as description_length
            FROM {$wpdb->posts} p
            INNER JOIN {$wpdb->postmeta} pm ON p.ID = pm.post_id 
                AND pm.meta_key = '_yoast_wpseo_metadesc'
            WHERE p.post_type IN ({$postTypesSql})
                AND p.post_status = 'publish'
                AND LENGTH(pm.meta_value) > %d
            ORDER BY LENGTH(pm.meta_value) DESC
        ", $maxLength), ARRAY_A);
    }
    
    public function findMissingCategoryKeywords(): array
    {
        global $wpdb;
        
        return $wpdb->get_results("
            SELECT t.term_id, t.name, t.slug, tt.taxonomy
            FROM {$wpdb->terms} t
            INNER JOIN {$wpdb->term_taxonomy} tt ON t.term_id = tt.term_id
            LEFT JOIN {$wpdb->termmeta} tm ON t.term_id = tm.term_id 
                AND tm.meta_key = 'wpseo_focuskw'
            WHERE tt.taxonomy = 'category'
                AND (tm.meta_value IS NULL OR tm.meta_value = '')
            ORDER BY t.name ASC
        ", ARRAY_A);
    }
    
    public function getContentStats(): array
    {
        return [
            'posts_missing_keyword' => count($this->findMissingFocusKeywords(['post_types' => ['post']])),
            'pages_missing_keyword' => count($this->findMissingFocusKeywords(['post_types' => ['page']])),
            'categories_missing_keyword' => count($this->findMissingCategoryKeywords()),
            'oversized_descriptions' => count($this->findOversizedMetaDescriptions(
                $this->settings->getSetting('meta_description.max_length')
            ))
        ];
    }
}
```

## Batch Optimization

### Queue Processing

```php
class YoastOptimizationProcessor
{
    public function processQueue(int $batchSize = 25): ProcessingResult
    {
        $items = $this->repository->getPendingItems($batchSize);
        $results = ['processed' => 0, 'success' => 0, 'failed' => 0];
        
        foreach ($items as $item) {
            try {
                $this->processItem($item);
                $item->status = OptimizationStatus::COMPLETED;
                $item->processedAt = new DateTimeImmutable();
                $results['success']++;
            } catch (Exception $e) {
                $item->status = OptimizationStatus::FAILED;
                $item->errorMessage = $e->getMessage();
                $results['failed']++;
            }
            
            $this->repository->save($item);
            $results['processed']++;
        }
        
        return new ProcessingResult($results);
    }
    
    private function processItem(YoastOptimizationQueueEntity $item): void
    {
        match($item->optimizationType) {
            OptimizationType::FOCUS_KEYWORD => $this->optimizeFocusKeyword($item->wpPostId),
            OptimizationType::MULTIPLE_KEYWORDS => $this->optimizeMultipleKeywords($item->wpPostId),
            OptimizationType::META_DESCRIPTION => $this->optimizeMetaDescription($item->wpPostId),
            default => throw new UnknownOptimizationTypeException(14951)
        };
    }
}
```
